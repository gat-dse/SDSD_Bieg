\newcommand{\RebarDots}[3]{%
    \foreach \x in {#1} {
        \fill[csRebar] (\x,#2) circle[radius=#3];
    }
}

\newcommand{\Dimension}[4]{%
    \draw[dim] (#1,#2) -- (#1,#3);
    \draw[dim] ({#1-0.06},#2) -- ({#1+0.06},#2);
    \draw[dim] ({#1-0.06},#3) -- ({#1+0.06},#3);
    \node[font=\small, rotate=90, anchor=south] at ({#1-0.11},{(#2+#3)/2}) {#4};
}

\newcommand{\LeftMarkers}[2]{%
    \draw[decorate, decoration={brace, amplitude=4pt}, dim]
        (-0.55,#1) -- (-0.55,#2)
        node[midway, left=5pt, note, text width=4.2cm] {
            Floor build-up\\
            generated automatically\\
            to fulfil acoustic requirements
        };
    \draw[decorate, decoration={brace, amplitude=4pt}, dim]
        (-0.55,0) -- (-0.55,#1)
        node[midway, left=5pt, note, text width=4.2cm] {
            Structural cross-section\\
            optimised for GWP\\
            with ULS, SLS deflection,\\
            SLS vibration and fire checks
        };
}

\newcommand{\RightParams}[1]{%
    \node[noteRight, anchor=west] at (4.05,1.25) {
        \textbf{Optimised parameters}\\[2pt]
        #1
    };
}

\newcommand{\FloorBuildUp}[2]{%
    % #1 = structural top y-coordinate, #2 = width
    \fill[csInsulation, mat] (0,#1) rectangle (#2,{#1+0.16});
    \fill[csScreed, mat] (0,{#1+0.16}) rectangle (#2,{#1+0.46});
    \fill[csParquet, mat] (0,{#1+0.46}) rectangle (#2,{#1+0.52});
    \node[label, anchor=west] at (#2+0.10,{#1+0.08}) {insulation};
    \node[label, anchor=west] at (#2+0.10,{#1+0.31}) {cement screed};
    \node[label, anchor=west] at (#2+0.10,{#1+0.49}) {parquet};
}

\newcommand{\SystemPage}[3]{%
    \clearpage
    \begin{center}
        {\Large\bfseries #1}\\[3mm]
        \begin{tikzpicture}[x=2.05cm,y=2.05cm]
            #2
            \RightParams{#3}
        \end{tikzpicture}
    \end{center}
}
